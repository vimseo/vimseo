    MockCurves(load_case_name: str = 'Dummy', **options)

    A toy model whose outputs illustrate the definition of figures holding a single line.

    Notes

    Since an IntegratedModel extends a GEMSEO <https://gemseo.readthedocs.io/en/stable/index.html>_ :class:~.gemseo.core.discipline.MDODiscipline, it can be used as |gemseo| discipline <https://gemseo.readthedocs.io/en/6.0.0/examples/api /plot_discipline.html#discipline>_. Also, |gemseo| capabilities (MLearning, Optimisation) can be used on the model.

    Examples:

    >>> # Run a model with default values, each execution being stored in a unique
>>> # directory. The archive and scratch persistency is customized.
>>> # The manager for archiving the results is set to MLFlow.
>>> from vimseo.api import create_model
>>> from vimseo.core.model_settings import IntegratedModelSettings
>>> model = create_model("BendingTestAnalytical", "Cantilever",
>>>     model_options=IntegratedModelSettings(
>>>         directory_archive_root=f"my_model_result",
>>>         directory_scratch_root=f"my_scratch",
>>>         directory_scratch_persistency=PersistencyPolicy.DELETE_ALWAYS,
>>>         directory_archive_persistency=PersistencyPolicy.DELETE_IF_FAULTY,
>>>         archive_manager="MlflowArchive",
>>>     ),
>>> )

    PLOTS: Sequence[tuple[str, ...]] = [
    ("x_history", "y_history")

    The figures to plot from the model data.

    A figure is defined either by a tuple of variable names, the first one being the abscissa and the following ones the ordinates, or by a :class:.Plot when the lines shall be styled, drawn against a secondary ordinate axis or completed with horizontal reference lines.

    A load case can override or extend these figures by declaring its own PLOTS class attribute. A load-case entry whose abscissa and first ordinate match a model entry replaces it in place; any other entry is appended. See :class:.LoadCase and :func:.merge_plots.

    Examples:

    >>> # A single line, and two lines sharing an abscissa.
>>> PLOTS = [
...     ("strain_history", "stress_history"),
...     ("time", "energy_strain_history", "energy_damage_history"),
... ]

    >>> # A styled figure with a secondary ordinate axis and a custom abscissa
>>> # label.
>>> PLOTS = [
...     Plot(
...         x="displacement_history",
...         x_label="Displacement",
...         traces=[
...             Trace("force_history", style=LineStyle(color="blue")),
...             Trace(
...                 "crack_position_history",

...                 style=LineStyle(color="red", marker="."),

    archive_outputs ¶
    archive_outputs(output_data)

    Archive the model result: field files, metadata, results, persistent files.

    Runs the archive half of the execute tail against the current scratch job directory — collecting field files (:meth:_collect_field_files), generating the metadata into output_data, then writing results.json and copying the persistent files through the archive manager. The archive job directory must already be created (self._archive_manager.create_job_directory()). Unlike :meth:_manage_persistency it does not touch the scratch-persistency policy, so a caller that ran individual disciplines (the vtt GUI's pre/run/post flow) can archive exactly as model.execute does without deleting the job directory it is reading.

    cls = self.__class__
    model_name = cls.__name__

        base for base in cls.__bases__ if issubclass(base, Discipline)
    ]
    names = [model_name] + [cls.__name__ for cls in classes[1:]]

    matching_files = []
    for cls, name in zip(classes, names, strict=False):
        class_module = sys.modules[cls.__module__]
        directory_path = Path(class_module.__file__).parent.absolute()
        file_path = directory_path / f"{name}{file_suffix}"
        if file_path.is_file():
            matching_files.append(file_path)
        for load_case in load_cases:
            file_path = directory_path / f"{name}_{load_case}{file_suffix}"
            if file_path.is_file():
                matching_files.append(file_path)

    if not matching_files:
        msg = (
            f"No files with suffix {file_suffix} were found next "
            f"to the model {self.name} or its parent class."
        )

    return matching_files

    {model_class_name}_{load_case}{file_suffix}.

    Args:
        file_suffix: The suffix, including the extension, of the file to be searched
            in next to model class or its parents.
        load_cases: an iterable of load cases. If specified, only the files placed
            next to these load case classes are searched.
            Otherwise, all load case classes associated to the model are considered.

    Returns:
        A list of file paths.
    """
    from vimseo.api import get_available_load_cases

    cls = self.__class__
    model_name = cls.__name__
    load_cases = load_cases or get_available_load_cases(model_name)
    classes = [cls] + [
        base for base in cls.__bases__ if issubclass(base, Discipline)
    ]
    names = [model_name] + [cls.__name__ for cls in classes[1:]]

    matching_files = []
    for cls, name in zip(classes, names, strict=False):
        class_module = sys.modules[cls.__module__]
        directory_path = Path(class_module.__file__).parent.absolute()
        file_path = directory_path / f"{name}{file_suffix}"
        if file_path.is_file():
            matching_files.append(file_path)
        for load_case in load_cases:
            file_path = directory_path / f"{name}_{load_case}{file_suffix}"
            if file_path.is_file():
                matching_files.append(file_path)

    if not matching_files:
        msg = (
            f"No files with suffix {file_suffix} were found next "
            f"to the model {self.name} or its parent class."
        )
        raise FileNotFoundError(msg)

    return matching_files

    self,
    directory_path: str | Path = "",
    save: bool = True,
    show: bool = False,
    scalar_names: Iterable[str] = (),
    data: str = "PLOTS",
) -> Mapping[str, Figure]:
    """Plots the results, from the definition of the figures in the PLOTS variable.

    Args:
        directory_path: A path where to save the plots, in files prefixed by the
            names of the model and of the load case. If empty, the job directory
            of the simulation in the archive, where the files are not prefixed
            (the path already names the model and the load case), else the
            current working directory.
        save: Whether to save the plot.
        show: Whether to show the plot.
        scalar_names: The scalars to plot in a scatter matrix. Show all scalars by default.
        data: Either ``"PLOTS"`` to draw the figures defined in ``PLOTS``,
            or ``"SCALARS"`` to draw a scatter matrix of the scalar outputs.

    Returns:
        The figures, indexed by their name.
    """
    # The job directory already names the model and the load case: a prefix
    # would only lengthen paths, which are limited to 260 characters on Windows.
    prefix = (
        f"{self.name}_{self._load_case.name}_"
        if directory_path != "" or not self.archive_manager.job_directory
        else ""
    )
    if save:
        directory_path = (
            Path(directory_path)
            if directory_path != ""
            else Path(self.archive_manager.job_directory or Path.cwd())
        )
        if not directory_path.exists():
            directory_path.mkdir(parents=True)
        LOGGER.info(f"Saving plots to {directory_path.absolute()}")
    else:
        directory_path = ""

    from vimseo.core.model_result import ModelResult

    result = ModelResult.from_dict(
        {"inputs": self.get_input_data(), "outputs": self.get_output_data()},
        model=self,
    )
    figures = {}
    if data == "PLOTS":
        figures = self._plot_curves(
            figures, result, directory_path, save, show, prefix
        )
    elif data == "SCALARS":
        plot = ScatterMatrix(
            Dataset.from_dataframe(
                DataFrame.from_dict([result.scalars(variable_names=scalar_names)])
            )
        )
        figures["scalars"] = plot.execute(
            directory_path=directory_path,
            file_name="scalars.html",
            save=save,
            show=show,
        )
    else:
        msg = f"Unknown data type {data} for plotting."
        raise ValueError(msg)

    return figures
